# Sir Sohail Agent: Actionable Advisory Prompts & Integration Guide

This guide contains **93 refined prompts (3 for each of your 31 variables)** designed to ask the **Sir Sohail Agent** for structured suggestions and recommendations based on the user's survey responses.

The survey responses map directly to three levels of implementation:
*   **Low Level (`Never` or `Rarely`):** Prompts focus on **how to start, introduce, or transition** to this strategy.
*   **Medium Level (`Sometimes` or `About half the time`):** Prompts focus on **how to optimize, structure, and improve consistency**.
*   **High Level (`Most of the time` or `Almost always`):** Prompts focus on **how to sustain, scale, and integrate** the strategy.

### 2. Processing Engine (`backend/users/services/analysis/agent_advisory_service.py`)

Create this file to calculate Likert weights and query the agent with fallback cache mechanics.

```python
import requests
import numpy as np
from django.conf import settings
from ...models import AgentRecommendationCache

LIKERT_WEIGHTS = {
    "Never": 1,
    "Rarely": 2,
    "Sometimes": 3,
    "About half the time": 4,
    "Most of the time": 5,
    "Almost always": 6
}

STRATEGY_MAP = {
    "Relevance 1p": "aligning course topics and materials with students' future career goals",
    "Relevance 2p": "providing real-world examples and practical applications of course concepts",
    "Relevance 3p": "explaining the importance, utility, and real-world value of topics taught",

    "Challenge 1p": "designing learning tasks that require critical analysis, synthesis, or evaluation",
    "Challenge 2p": "encouraging students to solve complex, open-ended problems without simple answers",
    "Challenge 3p": "setting high academic expectations and rigorous standards for student performance",
    "Challenge 4p": "assigning advanced readings or project work that stretches students' cognitive limits",
    "Challenge 5p": "prompting students to justify their reasoning and challenge assumptions during tasks",
    "Cncts 1p": "linking new course concepts to students' prior knowledge and academic experiences",
    "Cncts 2p": "integrating concepts from different subjects or disciplines into class activities",
    "Cncts 3p": "encouraging students to apply class learning to external personal or societal issues",
    "Cncts 4p": "connecting classroom learning with co-curricular or extracurricular experiences",
    "Cncts 5p": "helping students build cross-topic concepts and see the big picture of the curriculum"
}

def determine_level(score):
    if score < 2.5:
        return "Low"
    elif score < 4.5:
        return "Medium"
    else:
        return "High"

def get_recommendation_from_agent(column_name, level):
    cached_rec = AgentRecommendationCache.objects.filter(column_name=column_name, level=level).first()
    if cached_rec:
        return {
            "recommendation_text": cached_rec.recommendation_text,
            "citations": cached_rec.citations
        }

    strategy = STRATEGY_MAP.get(column_name, "")
    advisory_questions = {
        "Low": f"The current practice is implemented rarely or never. What suggestions and starting steps do the papers recommend to introduce or transition to: '{strategy}'? Provide a structured analysis of what works, what doesn't, and how to successfully start.",
        "Medium": f"The current practice is implemented sometimes or about half the time. How can teachers optimize and structure: '{strategy}' to make it more consistent? Provide suggestions on what works, what doesn't, and how to improve.",
        "High": f"The current practice is implemented most of the time or almost always. What suggestions exist to sustain student engagement, scale the delivery, or prevent fatigue for: '{strategy}'? Provide a structured analysis of what works, what doesn't, and how to scale/improve."
    }
    
    question = advisory_questions.get(level)
    prompt = f"""Based on the uploaded academic corpus, address the following advisory concern:
    
    "{question}"
    
    Instructions:
    - You must answer with a structured advisory analysis: what works, what doesn't, and how to improve.
    - Base all suggestions exclusively on the provided context in the PDFs. Do not use general knowledge or speculate.
    - Provide concrete quantitative findings or statistical outcomes verbatim where applicable.
    - Answer in a neutral, academic tone using concise bullets.
    - Every claim must be followed by a [N] page-level citation (no ranges like [1-3], list as [1][2][3])."""

    agent_url = "https://sir-sohail-agent.vercel.app/api/query"
    headers = {
        "Authorization": f"Bearer {settings.AGENT_API_SECRET}",
        "Content-Type": "application/json"
    }
    payload = {
        "message": prompt,
        "chatHistory": []
    }

    try:
        response = requests.post(agent_url, json=payload, headers=headers, timeout=45)
        response.raise_for_status()
        data = response.json()
        
        rec_text = data.get("answer", "No recommendations found.")
        citations = data.get("citations", [])

        AgentRecommendationCache.objects.create(
            column_name=column_name,
            level=level,
            recommendation_text=rec_text,
            citations=citations
        )

        return {"recommendation_text": rec_text, "citations": citations}
    except Exception as e:
        return {
            "recommendation_text": f"Agent service temporarily unavailable. (Detail: {str(e)})",
            "citations": []
        }

def analyze_class_wide_dataset(dataset_rows):
    results = {}
    for col in STRATEGY_MAP.keys():
        scores = [LIKERT_WEIGHTS[row[col]] for row in dataset_rows if row.get(col) in LIKERT_WEIGHTS]
        if scores:
            avg_score = np.mean(scores)
            level = determine_level(avg_score)
            advisory = get_recommendation_from_agent(col, level)
            results[col] = {
                "average_score": round(avg_score, 2),
                "level": level,
                "strategy": STRATEGY_MAP[col],
                "recommendation": advisory["recommendation_text"],
                "citations": advisory["citations"]
            }
    return results

def analyze_individual_students(dataset_rows):
    student_reports = []
    for idx, row in enumerate(dataset_rows):
        report = {"student_id": idx + 1, "profile": {}}
        for col in STRATEGY_MAP.keys():
            val = row.get(col)
            if val in LIKERT_WEIGHTS:
                numerical_score = LIKERT_WEIGHTS[val]
                level = determine_level(numerical_score)
                advisory = get_recommendation_from_agent(col, level)
                report["profile"][col] = {
                    "response": val,
                    "level": level,
                    "strategy": STRATEGY_MAP[col],
                    "recommendation": advisory["recommendation_text"],
                    "citations": advisory["citations"]
                }
        student_reports.append(report)
    return student_reports
```

### 3. API View Router (`backend/users/views.py`)

Add this endpoint inside your Django views file.

```python
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework import status
from rest_framework.permissions import IsAuthenticated
from .models import Dataset, AnalysisResult
from .services.analysis.agent_advisory_service import analyze_class_wide_dataset, analyze_individual_students

class AgentAdvisoryView(APIView):
    permission_classes = [IsAuthenticated]

    def post(self, request):
        dataset_id = request.data.get("dataset_id")
        analysis_mode = request.data.get("mode", "class-wide")  # 'class-wide' or 'individual'

        if not dataset_id:
            return Response({"error": "Missing dataset_id"}, status=status.HTTP_400_BAD_REQUEST)

        try:
            dataset = Dataset.objects.get(id=dataset_id, user=request.user)
            rows = dataset.cleaned_data
            
            if analysis_mode == "class-wide":
                results = analyze_class_wide_dataset(rows)
            else:
                results = analyze_individual_students(rows)

            AnalysisResult.objects.create(
                user=request.user,
                dataset=dataset,
                analysis_type=f'agent_{analysis_mode}',
                input_params={"analysis_mode": analysis_mode},
                output_results=results
            )

            return Response(results, status=status.HTTP_200_OK)

        except Dataset.DoesNotExist:
            return Response({"error": "Dataset not found"}, status=status.HTTP_404_NOT_FOUND)
        except Exception as e:
            return Response({"error": str(e)}, status=status.HTTP_500_INTERNAL_SERVER_ERROR)
```
